Task 2: Export Mumbai Indians Matches

SELECT match_id, season, city, date, team1, team2, toss_winner, toss_decision, winner, result_margin
FROM ipl_matches
WHERE team1 = 'Mumbai Indians' OR team2 = 'Mumbai Indians'
ORDER BY date ASC;

Task 3: Total Matches Played Per Team (Excel Model)

WITH AllTeams AS (
    SELECT team1 AS team FROM ipl_matches
    UNION ALL
    SELECT team2 AS team FROM ipl_matches
)
SELECT team, COUNT(*) AS total_matches_played
FROM AllTeams
GROUP BY team
ORDER BY total_matches_played DESC;

import pandas as pd
import sqlite3

# Load CSV
df = pd.read_csv("mi_matches.csv")

# Filter where Mumbai Indians won
mi_wins_df = df[df["winner"] == "Mumbai Indians"]

# Insert into new SQL table mi_wins
conn = sqlite3.connect("session_20_ipl.db")
mi_wins_df.to_sql("mi_wins", conn, if_exists="replace", index=False)
conn.close()
