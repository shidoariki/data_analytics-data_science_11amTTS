2. Assignment Solutions & M-Scripts

Task 2: Cell-Bound Parameterized Power Query (M-Code)

let
    // 1. Fetch value from Excel named cell 'SelectedCuisine'
    CuisineParam = Excel.CurrentWorkbook(){[Name="SelectedCuisine"]}[Content]{0}[Column1],

    // 2. Execute SQL query with parameter pushdown
    Source = Sql.Database("localhost", "analytics_db", [
        Query = "SELECT name, cuisine, rating, cost_for_two, city FROM Restaurants WHERE cuisine = '" & Text.Replace(CuisineParam, "'", "''") & "' ORDER BY rating DESC"
    ])

Task 3: Dynamic Data Refresh Configuration

  • Automatic Background Refresh: Configured in Connection Properties under Usage.
  • Open File Refresh: Automatically syncs latest SQL state upon opening the workbook.
  • One-Click Sync: Ctrl + Alt + F5 ("Refresh All") executes immediate re-query.

Task 4: High Rated Restaurants (> 4.0) SQL

SELECT name, cuisine, rating, cost_for_two, city
FROM Restaurants
WHERE rating > 4.0
ORDER BY rating DESC;

Task 5: Range-Bound Parameterized M-Code

let
    MinR = Text.From(Excel.CurrentWorkbook(){[Name="MinRating"]}[Content]{0}[Column1]),
    MaxR = Text.From(Excel.CurrentWorkbook(){[Name="MaxRating"]}[Content]{0}[Column1]),
    Source = Sql.Database("localhost", "analytics_db", [
        Query = "SELECT name, cuisine, rating, cost_for_two, city FROM Restaurants WHERE rating BETWEEN " & MinR & " AND " & MaxR & " ORDER BY rating DESC"
